# app/db/engine.py

Async-движок поверх aiosqlite. init_db() создаёт parent dir, открывает соединение, applies schema.sql через executescript, потом _apply_migrations. _apply_migrations выполняет два набора миграций: (1) идемпотентные ALTER TABLE (try/except 'duplicate column'/'already exists'); текущее: subscriptions.xui_sub_id TEXT NOT NULL DEFAULT ''. (2) CREATE TABLE/INDEX IF NOT EXISTS для таблиц, появившихся после первоначальной схемы; текущее: subscription_notifications + idx_subscription_notifications_sub. _configure_connection ставит row_factory=Row, PRAGMA foreign_keys=ON и journal_mode=WAL (per-connection). get_conn() async ctxmgr открывает/конфигурирует/закрывает соединение. transaction(conn=None) async ctxmgr оборачивает BEGIN IMMEDIATE/COMMIT/ROLLBACK блок (для критических секций типа promo redemption).
