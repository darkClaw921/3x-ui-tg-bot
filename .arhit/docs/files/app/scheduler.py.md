# app/scheduler.py

APScheduler фоновые задачи бота. Создаёт AsyncIOScheduler (timezone=UTC) с тремя cron-job-ами: expire_check_job (cron minute=0, ежечасно) — переводит истёкшие подписки в 'expired', дисейблит клиента через xui.update_client(enable=False), шлёт юзеру финальное уведомление (дедуп через subscription_notifications kind='expired'); reminders_job (cron hour=10 minute=0, раз в сутки) — для каждой подписки в окне 3 дней через _days_left + _kind_for_days_left маппит остаток в 3d/1d/0d и шлёт соответствующее сообщение из _REMINDER_TEXTS (дедуп через try_mark_notification_sent); traffic_snapshot_job (cron hour=0,6,12,18 minute=5) — для каждой active подписки вызывает xui.get_client_traffics(email) и пишет в traffic_snapshots. Все job-ы устойчивы к ошибкам отдельных подписок (try/except + logger.exception в цикле). setup_scheduler(bot) -> AsyncIOScheduler возвращает настроенный scheduler без старта; вызывает его из app/main.py через scheduler.start()/scheduler.shutdown(wait=False). Bot инстанс прокидывается через closure _wrap(job, bot, name). Зависит от: app.db.engine, app.db.repos.subscriptions, app.db.repos.users, app.xui.get_xui_client, app.xui.clients.update_client/get_client_traffics.
