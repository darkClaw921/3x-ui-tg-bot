# app/xui/links.py

Билдеры ссылок и QR. Функции: build_subscription_url(sub_id)->str — джойнит settings.XUI_SUB_BASE_URL с sub_id, нормализуя слеши; build_vless_link(inbound, client_uuid, email)->str — собирает vless://uuid@host:port?<params>#<email> URI на основе распарсенного inbound (см. app.xui.inbounds.get_inbound); host берётся из settings.XUI_SERVER_HOST (не из panel), port из inbound.port; query-параметры извлекаются из streamSettings через _stream_params: type/security всегда; для tcp — headerType/path/host (из header.request), ws — path/host, grpc — serviceName/mode, http — path/host, kcp — headerType/seed, quic — quicSecurity/key/headerType; для tls — sni/alpn/fp; для reality — pbk/fp/sni/sid/spx; flow добавляется из client object (settings.clients[i].flow); _find_client ищет клиент по uuid в inbound.settings.clients; fragment — URL-encoded email через urllib.parse.quote(safe=''); make_qr_png(text)->bytes — qrcode.make + сохранение в BytesIO как PNG; возвращает байты, готовые для send_photo. Зависимости: qrcode[pil], urllib.parse, io.BytesIO, app.config.settings (XUI_SERVER_HOST, XUI_SUB_BASE_URL).
